\newcommand{\regularpolygon}[4]{%
  \draw[#4,line width=0.8pt]
  \foreach \k in {0,...,#1} {
    \ifnum\k=0
      ($(#2)+({#3*cos(90+360*\k/#1)},{#3*sin(90+360*\k/#1)})$)
    \else
      -- ($(#2)+({#3*cos(90+360*\k/#1)},{#3*sin(90+360*\k/#1)})$)
    \fi
  };
}
