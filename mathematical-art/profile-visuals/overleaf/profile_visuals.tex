\newcommand{\regpoly}[5]{%
  \draw[#5,line join=round]
  \foreach \j in {0,...,#1} {
    \ifnum\j=0
      ($(#2)+({#4*cos(90+360*\j/#1)},{#4*sin(90+360*\j/#1)})$)
    \else
      -- ($(#2)+({#4*cos(90+360*\j/#1)},{#4*sin(90+360*\j/#1)})$)
    \fi
  };
}
