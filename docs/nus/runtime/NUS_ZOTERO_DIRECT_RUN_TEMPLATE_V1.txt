function fail(ok, code, detail = null) {
  if (!ok) {
    throw new Error(
      "FAIL:" + code + (detail === null ? "" : ":" + JSON.stringify(detail))
    );
  }
}

async function maybeAwait(x) {
  return await Promise.resolve(x);
}

async function sha256Bytes(bytes) {
  const digest = await globalThis.crypto.subtle.digest("SHA-256", bytes);
  return [...new Uint8Array(digest)]
    .map(x => x.toString(16).padStart(2, "0"))
    .join("");
}

async function sha256File(path) {
  return sha256Bytes(await IOUtils.read(path));
}

async function getLiveAnnotations(attachment) {
  const annotations = await maybeAwait(attachment.getAnnotations?.() || []);
  fail(Array.isArray(annotations), "ANNOTATION_COLLECTION_NOT_ARRAY");
  return annotations.filter(a => a && !a.deleted);
}

function makeComment(record) {
  const exact = String(record.exact_author_text || "");
  switch (record.role) {
    case "PURPOSE":
      return `Purpose : ${exact}`;
    case "INTERVENTION":
      return `Intervention : ${exact}`;
    case "INPUT":
      return `Input : ${exact}`;
    case "ENGINEERING_METHOD":
      return `Engineering method : ${exact}`;
    case "ENGINEERING_OUTPUT":
      fail(ENGINEERING_NATURES.has(record.output_nature), "ENGINEERING_NATURE_INVALID");
      return `Engineering output here is ${record.output_nature} : ${exact}`;
    case "SUSTAINABILITY_TRANSFORMATION":
      return `Sustainability transformation : ${exact}`;
    case "SUSTAINABILITY_METHOD":
      return `Sustainability method : ${exact}`;
    case "SUSTAINABLE_OUTCOME":
      fail(SUSTAINABILITY_NATURES.has(record.output_nature), "SUSTAINABILITY_NATURE_INVALID");
      return `Sustainability output here is ${record.output_nature} : ${exact}`;
    case "DECISION":
      return `Decision : ${exact}`;
    default:
      fail(false, "UNKNOWN_ROLE", record.role);
  }
}

async function resolvePaperIdentity(paper) {
  fail(paper, "PAPER_CONFIG_MISSING");
  fail(paper.parent_key && paper.attachment_key, "PAPER_IDENTITY_INCOMPLETE", paper.paper_id);

  const libraryID = Zotero.Libraries.userLibraryID;
  const parent = await maybeAwait(
    Zotero.Items.getByLibraryAndKey(libraryID, paper.parent_key)
  );
  const attachment = await maybeAwait(
    Zotero.Items.getByLibraryAndKey(libraryID, paper.attachment_key)
  );

  fail(parent && attachment, "IDENTITY_NOT_FOUND", paper.paper_id);
  fail(
    attachment.parentID === parent.id || attachment.parentItemID === parent.id,
    "PARENT_ATTACHMENT_MISMATCH",
    paper.paper_id
  );

  const pdfPath = await attachment.getFilePathAsync();
  fail(pdfPath && await IOUtils.exists(pdfPath), "PDF_FILE_MISSING", paper.paper_id);

  const actualSha = await sha256File(pdfPath);
  fail(actualSha === paper.pdf_sha256, "PDF_SHA256_MISMATCH", {
    paper_id: paper.paper_id,
    expected: paper.pdf_sha256,
    actual: actualSha
  });

  return { parent, attachment, pdfPath, pdf_sha256: actualSha };
}

async function getReader(attachment) {
  let reader = await Zotero.Reader.open(attachment.id);
  if (!reader) {
    reader = Zotero.Reader._readers.find(
      x => x?._item?.id === attachment.id && !x._isTabClosed
    );
  }
  fail(reader, "READER_OPEN_FAILED");
  await reader._initPromise;
  fail(reader._internalReader, "INTERNAL_READER_MISSING");
  await reader._internalReader._primaryView.initializedPromise;
  return reader;
}

function assertNativeAnnotationAPI(reader) {
  const manager = reader?._internalReader?._annotationManager;
  fail(manager, "NATIVE_ANNOTATION_MANAGER_UNAVAILABLE");
  fail(typeof manager.addAnnotation === "function", "NATIVE_ADD_ANNOTATION_API_UNAVAILABLE");
  return manager;
}

// Keep every Reader-internal annotation creation call in this adapter only.
async function createNativeAnnotation(reader, record, sortIndex) {
  const manager = assertNativeAnnotationAPI(reader);

  const raw = {
    type: record.zotero_annotation_type,
    color: record.annotation_color,
    sortIndex,
    pageLabel: String(record.page_label),
    position: record.position,
    comment: record.annotation_comment,
    tags: []
  };

  if (record.zotero_annotation_type === "highlight") {
    raw.text = record.annotation_text;
  }

  const created = manager.addAnnotation(
    Components.utils.cloneInto(raw, reader._iframeWindow)
  );

  fail(created?.id, "NATIVE_CREATE_NO_ID", record.schema_id);
  return created;
}

async function runDiscover() {
  fail(TARGET_PAPER_ID !== null, "TARGET_PAPER_ID_REQUIRED");
  const paper = PAPERS[TARGET_PAPER_ID];
  fail(paper, "TARGET_PAPER_CONFIG_NOT_EMBEDDED", TARGET_PAPER_ID);
  const identity = await resolvePaperIdentity(paper);
  const live = await getLiveAnnotations(identity.attachment);
  return {
    protocol: "NUS_ZOTERO_DIRECT_RUN_TEMPLATE_V1",
    mode: MODE,
    status: "PASS_DISCOVER_TEMPLATE",
    paper_id: TARGET_PAPER_ID,
    pdf_sha256: identity.pdf_sha256,
    live_annotation_count: live.length,
    mutation_performed: false
  };
}

async function run() {

  switch (MODE) {
    case "STATUS":
      return await runStatus();
    case "DISCOVER":
      return await runDiscover();
    case "PREFLIGHT":
      return await runPreflight();
    case "AUDIT":
      return await runAudit();
    case "WRITE":
      return await runWrite();

    case "ADJUDICATE_CHECK":
    case "GEOMETRY":
    case "FREEZE_CHECK":
      fail(false, "MODE_NOT_IMPLEMENTED_IN_TEMPLATE", MODE);
    default:
      fail(false, "UNKNOWN_MODE", MODE);
  }
}
